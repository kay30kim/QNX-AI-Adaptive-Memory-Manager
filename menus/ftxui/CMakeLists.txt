cmake_minimum_required(VERSION 3.20)
project(qnx_menu_ftxui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(ftxui GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git GIT_TAG v6.1.9 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(ftxui)
add_executable(qnx-menu-ftxui src/main.cpp src/engine.cpp)
target_include_directories(qnx-menu-ftxui PRIVATE include)
target_link_libraries(qnx-menu-ftxui PRIVATE ftxui::screen ftxui::dom ftxui::component)
if(MINGW)
  target_link_options(qnx-menu-ftxui PRIVATE -static)
endif()
